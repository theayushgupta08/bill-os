## 8.1 Design Principles

| Principle | Implementation |
|-----------|---------------|
| **UUID Primary Keys** | All tables use `UUID` as PK (generated by `gen_random_uuid()`) |
| **Soft Deletes** | `deleted_at TIMESTAMP NULL` — never physically delete business data |
| **Optimistic Locking** | `version INTEGER DEFAULT 0` — prevents concurrent update conflicts |
| **JPA Auditing** | `created_at`, `updated_at`, `created_by`, `updated_by` on all tables |
| **Tenant Isolation** | `organization_id UUID NOT NULL` on all business tables |
| **Normalized** | 3NF minimum; denormalize only for proven performance bottlenecks |
| **Flyway Only** | All schema changes via numbered migrations — never manual DDL |
| **Indian Compliance** | Fields for GSTIN, HSN/SAC, PAN, Indian states, tax breakups |

## 8.3 Complete Table Definitions

### 8.3.1 `users`

> System-wide user accounts (not tenant-scoped).

```sql
CREATE TABLE users (
    id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email             VARCHAR(255) NOT NULL,
    password_hash     VARCHAR(255) NOT NULL,
    full_name         VARCHAR(200) NOT NULL,
    phone             VARCHAR(15),
    avatar_url        VARCHAR(500),
    status            VARCHAR(20) NOT NULL DEFAULT 'UNVERIFIED',
        -- UNVERIFIED, ACTIVE, SUSPENDED, DEACTIVATED
    email_verified_at TIMESTAMP WITH TIME ZONE,
    last_login_at     TIMESTAMP WITH TIME ZONE,
    failed_login_count INTEGER NOT NULL DEFAULT 0,
    locked_until      TIMESTAMP WITH TIME ZONE,
    created_at        TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    updated_at        TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    version           INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT uq_users_email UNIQUE (email),
    CONSTRAINT ck_users_status CHECK (status IN ('UNVERIFIED','ACTIVE','SUSPENDED','DEACTIVATED'))
);

CREATE INDEX idx_users_email ON users (email);
CREATE INDEX idx_users_status ON users (status);
```

### 8.3.2 `organizations`

> Tenant / company records.

```sql
CREATE TABLE organizations (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(200) NOT NULL,
    legal_name          VARCHAR(200) NOT NULL,
    business_type       VARCHAR(30) NOT NULL,
        -- SOLE_PROPRIETORSHIP, PARTNERSHIP, LLP, PVT_LTD, PUBLIC_LTD, TRUST, SOCIETY, OTHER
    industry            VARCHAR(30),
    logo_url            VARCHAR(500),

    -- GST Details
    gstin               VARCHAR(15),
    gst_registration_type VARCHAR(20) NOT NULL DEFAULT 'REGULAR',
        -- REGULAR, COMPOSITION, UNREGISTERED, CONSUMER, OVERSEAS, SEZ
    trade_name          VARCHAR(200),
    pan                 VARCHAR(10),
    state_code          VARCHAR(2),
    gst_registration_date DATE,
    einvoice_enabled    BOOLEAN NOT NULL DEFAULT false,
    eway_bill_enabled   BOOLEAN NOT NULL DEFAULT false,

    -- Address
    address_line1       VARCHAR(200) NOT NULL,
    address_line2       VARCHAR(200),
    city                VARCHAR(100) NOT NULL,
    state               VARCHAR(50) NOT NULL,
    pin_code            VARCHAR(6) NOT NULL,
    country             VARCHAR(50) NOT NULL DEFAULT 'India',

    -- Contact
    phone               VARCHAR(15) NOT NULL,
    email               VARCHAR(255) NOT NULL,
    website             VARCHAR(255),

    -- Financial Year
    fy_start_month      INTEGER NOT NULL DEFAULT 4,  -- April
    current_fy_start    DATE NOT NULL,
    current_fy_end      DATE NOT NULL,

    -- Settings (JSONB for flexible configuration)
    settings            JSONB NOT NULL DEFAULT '{}',

    -- Audit
    created_by          UUID REFERENCES users(id),
    created_at          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    updated_at          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    deleted_at          TIMESTAMP WITH TIME ZONE,
    version             INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT ck_org_business_type CHECK (
        business_type IN ('SOLE_PROPRIETORSHIP','PARTNERSHIP','LLP','PVT_LTD','PUBLIC_LTD','TRUST','SOCIETY','OTHER')
    )
);

CREATE INDEX idx_org_gstin ON organizations (gstin) WHERE gstin IS NOT NULL;
CREATE INDEX idx_org_deleted ON organizations (deleted_at);
```

### 8.3.3 `user_org_roles`

> Maps users to organizations with roles. Enables multi-org membership.

```sql
CREATE TABLE user_org_roles (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    organization_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    role            VARCHAR(20) NOT NULL,
        -- OWNER, ADMIN, MANAGER, ACCOUNTANT, STAFF
    branch_ids      UUID[],  -- Optional: restrict to specific branches
    invited_by      UUID REFERENCES users(id),
    invited_at      TIMESTAMP WITH TIME ZONE,
    accepted_at     TIMESTAMP WITH TIME ZONE,
    status          VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
        -- PENDING, ACTIVE, SUSPENDED
    created_at      TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    updated_at      TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),

    CONSTRAINT uq_user_org UNIQUE (user_id, organization_id),
    CONSTRAINT ck_uor_role CHECK (role IN ('OWNER','ADMIN','MANAGER','ACCOUNTANT','STAFF')),
    CONSTRAINT ck_uor_status CHECK (status IN ('PENDING','ACTIVE','SUSPENDED'))
);

CREATE INDEX idx_uor_user ON user_org_roles (user_id);
CREATE INDEX idx_uor_org ON user_org_roles (organization_id);
```

### 8.3.6 `customers`

```sql
CREATE TABLE customers (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    organization_id UUID NOT NULL REFERENCES organizations(id),
    customer_code   VARCHAR(50) NOT NULL,
    display_name    VARCHAR(200) NOT NULL,
    legal_name      VARCHAR(200),
    customer_type   VARCHAR(20) NOT NULL DEFAULT 'BUSINESS',
        -- BUSINESS, INDIVIDUAL
    gstin           VARCHAR(15),
    pan             VARCHAR(10),
    email           VARCHAR(255),
    phone           VARCHAR(15),

    -- Billing Address
    billing_address_line1  VARCHAR(200),
    billing_address_line2  VARCHAR(200),
    billing_city           VARCHAR(100),
    billing_state          VARCHAR(50),
    billing_state_code     VARCHAR(2),
    billing_pin_code       VARCHAR(6),

    -- Shipping Address (if different)
    shipping_address_line1 VARCHAR(200),
    shipping_address_line2 VARCHAR(200),
    shipping_city          VARCHAR(100),
    shipping_state         VARCHAR(50),
    shipping_state_code    VARCHAR(2),
    shipping_pin_code      VARCHAR(6),

    -- Financial
    credit_limit    DECIMAL(15,2) NOT NULL DEFAULT 0,
    payment_terms   INTEGER NOT NULL DEFAULT 30,
    opening_balance DECIMAL(15,2) NOT NULL DEFAULT 0,
    outstanding     DECIMAL(15,2) NOT NULL DEFAULT 0,  -- Computed, updated on transactions

    notes           TEXT,
    tags            TEXT[],
    status          VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',

    -- Audit
    created_by      UUID REFERENCES users(id),
    created_at      TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    updated_at      TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    deleted_at      TIMESTAMP WITH TIME ZONE,
    version         INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT uq_customer_code UNIQUE (organization_id, customer_code),
    CONSTRAINT ck_customer_credit CHECK (credit_limit >= 0)
);

CREATE INDEX idx_customer_org ON customers (organization_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_customer_gstin ON customers (organization_id, gstin) WHERE gstin IS NOT NULL AND deleted_at IS NULL;
CREATE INDEX idx_customer_name ON customers USING gin (display_name gin_trgm_ops);
CREATE INDEX idx_customer_phone ON customers (organization_id, phone) WHERE deleted_at IS NULL;
```

### 8.3.9 `products`

```sql
CREATE TABLE products (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    organization_id     UUID NOT NULL REFERENCES organizations(id),
    product_code        VARCHAR(50) NOT NULL,
    name                VARCHAR(200) NOT NULL,
    description         TEXT,
    product_type        VARCHAR(10) NOT NULL DEFAULT 'GOODS',
        -- GOODS, SERVICE
    category_id         UUID REFERENCES product_categories(id),
    brand               VARCHAR(100),

    -- Tax
    hsn_code            VARCHAR(8),
    sac_code            VARCHAR(8),
    gst_rate            DECIMAL(5,2) NOT NULL DEFAULT 18.00,
    cess_rate           DECIMAL(5,2) NOT NULL DEFAULT 0,
    tax_inclusive        BOOLEAN NOT NULL DEFAULT false,

    -- Pricing
    purchase_price      DECIMAL(15,2) NOT NULL DEFAULT 0,
    selling_price       DECIMAL(15,2) NOT NULL DEFAULT 0,
    mrp                 DECIMAL(15,2),
    min_selling_price   DECIMAL(15,2),
    wholesale_price     DECIMAL(15,2),

    -- Unit
    unit_of_measure     VARCHAR(10) NOT NULL DEFAULT 'PCS',

    -- Inventory settings
    track_inventory     BOOLEAN NOT NULL DEFAULT true,
    reorder_level       INTEGER,
    reorder_quantity    INTEGER,
    batch_tracking      BOOLEAN NOT NULL DEFAULT false,
    expiry_tracking     BOOLEAN NOT NULL DEFAULT false,

    -- Media
    images              TEXT[],  -- Array of storage URLs

    status              VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    created_by          UUID REFERENCES users(id),
    created_at          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    updated_at          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    deleted_at          TIMESTAMP WITH TIME ZONE,
    version             INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT uq_product_code UNIQUE (organization_id, product_code),
    CONSTRAINT ck_product_type CHECK (product_type IN ('GOODS','SERVICE')),
    CONSTRAINT ck_gst_rate CHECK (gst_rate IN (0, 0.25, 3, 5, 12, 18, 28)),
    CONSTRAINT ck_prices CHECK (purchase_price >= 0 AND selling_price >= 0)
);

CREATE INDEX idx_product_org ON products (organization_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_product_category ON products (category_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_product_hsn ON products (organization_id, hsn_code) WHERE deleted_at IS NULL;
CREATE INDEX idx_product_name ON products USING gin (name gin_trgm_ops);
```

## 8.4 Views

### Outstanding Receivables View

```sql
CREATE VIEW v_outstanding_receivables AS
SELECT
    si.organization_id,
    si.customer_id,
    c.display_name AS customer_name,
    si.id AS invoice_id,
    si.invoice_number,
    si.invoice_date,
    si.due_date,
    si.total_amount,
    si.paid_amount,
    si.due_amount,
    CASE
        WHEN CURRENT_DATE - si.due_date <= 0 THEN 'CURRENT'
        WHEN CURRENT_DATE - si.due_date BETWEEN 1 AND 30 THEN '1-30 DAYS'
        WHEN CURRENT_DATE - si.due_date BETWEEN 31 AND 60 THEN '31-60 DAYS'
        WHEN CURRENT_DATE - si.due_date BETWEEN 61 AND 90 THEN '61-90 DAYS'
        ELSE '90+ DAYS'
    END AS aging_bucket
FROM sales_invoices si
JOIN customers c ON si.customer_id = c.id
WHERE si.status NOT IN ('DRAFT', 'VOID', 'CANCELLED', 'PAID')
  AND si.deleted_at IS NULL
  AND si.due_amount > 0;
```

### Daily Sales Summary View

```sql
CREATE VIEW v_daily_sales_summary AS
SELECT
    si.organization_id,
    si.invoice_date,
    COUNT(*) AS invoice_count,
    SUM(si.subtotal) AS total_subtotal,
    SUM(si.tax_amount) AS total_tax,
    SUM(si.total_amount) AS total_amount,
    SUM(si.paid_amount) AS total_paid,
    SUM(si.due_amount) AS total_due
FROM sales_invoices si
WHERE si.status NOT IN ('DRAFT', 'VOID', 'CANCELLED')
  AND si.deleted_at IS NULL
GROUP BY si.organization_id, si.invoice_date;
```

### Update `updated_at` Trigger

CREATE OR REPLACE FUNCTION fn_update_timestamp()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

### Update Customer Outstanding Trigger

```sql
CREATE OR REPLACE FUNCTION fn_update_customer_outstanding()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE customers
    SET outstanding = (
        SELECT COALESCE(SUM(due_amount), 0)
        FROM sales_invoices
        WHERE customer_id = COALESCE(NEW.customer_id, OLD.customer_id)
          AND organization_id = COALESCE(NEW.organization_id, OLD.organization_id)
          AND status NOT IN ('DRAFT', 'VOID', 'CANCELLED')
          AND deleted_at IS NULL
    )
    WHERE id = COALESCE(NEW.customer_id, OLD.customer_id);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_invoice_outstanding
    AFTER INSERT OR UPDATE OF paid_amount, due_amount, status ON sales_invoices
    FOR EACH ROW EXECUTE FUNCTION fn_update_customer_outstanding();
```
